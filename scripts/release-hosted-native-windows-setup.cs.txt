// Fixed setup primitives for the disposable hosted native account. No credential
// store, arbitrary command, account adoption, deletion or acceptance authority.
using System;
using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Security;
using System.Security.AccessControl;
using System.Security.Cryptography;
using System.Security.Principal;
using System.Text;

namespace Portcove.HostedNative {
    public static class Setup {
        [StructLayout(LayoutKind.Sequential)] struct Attributes {
            public int length; public IntPtr descriptor; public int inherit;
        }
        [DllImport("kernel32.dll", CharSet=CharSet.Unicode, SetLastError=true)]
        static extern bool CreateDirectoryW(string path, ref Attributes attributes);
        [DllImport("userenv.dll", CharSet=CharSet.Unicode)]
        static extern int CreateProfile(string sid, string name, StringBuilder path, uint size);

        public static SecureString Password() {
            byte[] bytes = new byte[32]; RandomNumberGenerator.Fill(bytes);
            var value = new SecureString();
            try {
                foreach (char c in "Pc4!") value.AppendChar(c);
                foreach (byte b in bytes) {
                    value.AppendChar("0123456789abcdef"[b >> 4]);
                    value.AppendChar("0123456789abcdef"[b & 15]);
                }
                value.MakeReadOnly(); return value;
            } catch { value.Dispose(); throw; }
            finally { Array.Clear(bytes, 0, bytes.Length); }
        }
        public static string Profile(string sid) {
            // This fixed short name uses the successfully qualified 260-character
            // buffer, not the earlier failed 32768-character setup call.
            var result = new StringBuilder(260);
            int status = CreateProfile(sid, "PortcoveNativeQual", result, 260);
            if (status != 0) Marshal.ThrowExceptionForHR(status);
            return result.ToString();
        }
        public static void Directory(string path, string owner, string candidate, bool writable) {
            var acl = new DirectorySecurity();
            acl.SetOwner(new SecurityIdentifier(owner));
            acl.SetAccessRuleProtection(true, false);
            var inheritance = InheritanceFlags.ContainerInherit | InheritanceFlags.ObjectInherit;
            foreach (string sid in new [] {owner, "S-1-5-18"})
                acl.AddAccessRule(new FileSystemAccessRule(new SecurityIdentifier(sid),
                    FileSystemRights.FullControl, inheritance, PropagationFlags.None, AccessControlType.Allow));
            if (!String.IsNullOrEmpty(candidate))
                acl.AddAccessRule(new FileSystemAccessRule(new SecurityIdentifier(candidate),
                    writable ? FileSystemRights.Modify : FileSystemRights.ReadAndExecute,
                    inheritance, PropagationFlags.None, AccessControlType.Allow));
            byte[] bytes = acl.GetSecurityDescriptorBinaryForm();
            var pinned = GCHandle.Alloc(bytes, GCHandleType.Pinned);
            try {
                var attributes = new Attributes {length=Marshal.SizeOf<Attributes>(), descriptor=pinned.AddrOfPinnedObject()};
                if (!CreateDirectoryW(path, ref attributes))
                    throw new Win32Exception(Marshal.GetLastWin32Error(), "Fresh private native directory required; no adoption");
            } finally { pinned.Free(); }
        }
    }
    // Hosted-only proposal: existing station access is a reviewed authority delta.
    public sealed class HostedDesktop : IDisposable {
        public const bool ExecutionAuthorized = true;
        public string Name { get; private set; }
        IntPtr station, desktop;
        byte[] before, applied;
        bool changed;
        public bool Restored { get; private set; }
        [DllImport("user32", CharSet=CharSet.Unicode, SetLastError=true)] static extern IntPtr OpenWindowStationW(string name, bool inherit, uint access);
        [DllImport("user32", SetLastError=true)] static extern IntPtr GetProcessWindowStation();
        [DllImport("user32", SetLastError=true)] static extern bool SetProcessWindowStation(IntPtr station);
        [DllImport("user32", CharSet=CharSet.Unicode, SetLastError=true)] static extern IntPtr OpenDesktopW(string name, uint flags, bool inherit, uint access);
        [DllImport("user32", CharSet=CharSet.Unicode, SetLastError=true)] static extern IntPtr CreateDesktopW(string name, IntPtr device, IntPtr mode, uint flags, uint access, ref Attributes attributes);
        [DllImport("user32", SetLastError=true)] static extern bool CloseDesktop(IntPtr desktop);
        [DllImport("user32", SetLastError=true)] static extern bool CloseWindowStation(IntPtr station);
        [DllImport("user32", SetLastError=true)] static extern bool GetUserObjectSecurity(IntPtr handle, ref uint information, byte[] descriptor, uint length, out uint needed);
        [DllImport("user32", SetLastError=true)] static extern bool SetUserObjectSecurity(IntPtr handle, ref uint information, byte[] descriptor);
        [DllImport("user32", CharSet=CharSet.Unicode, SetLastError=true)] static extern bool GetUserObjectInformationW(IntPtr handle, int index, out int value, uint length, out uint needed);
        [StructLayout(LayoutKind.Sequential)] struct Attributes { public int length; public IntPtr descriptor; public int inherit; }
        static byte[] ReadAcl(IntPtr handle) {
            uint information = 7, needed;
            GetUserObjectSecurity(handle, ref information, null, 0, out needed);
            if (needed < 20 || needed > 8192) throw new InvalidOperationException("User-object security exceeds bound");
            byte[] bytes = new byte[needed];
            if (!GetUserObjectSecurity(handle, ref information, bytes, needed, out needed)) throw new Win32Exception(Marshal.GetLastWin32Error(), "Read object DACL");
            return bytes;
        }
        static void WriteAcl(IntPtr handle, byte[] bytes) {
            uint information = 4;
            if (!SetUserObjectSecurity(handle, ref information, bytes)) throw new Win32Exception(Marshal.GetLastWin32Error(), "Write object DACL");
        }
        public static byte[] StationProposal(byte[] original, string candidate) {
            var descriptor = new RawSecurityDescriptor(original, 0);
            if (descriptor.DiscretionaryAcl == null) throw new InvalidOperationException("Null station DACL refused");
            var sid = new SecurityIdentifier(candidate);
            foreach (GenericAce ace in descriptor.DiscretionaryAcl) {
                var known = ace as KnownAce;
                if (known != null && known.SecurityIdentifier.Equals(sid)) throw new InvalidOperationException("Existing candidate station ACE refused");
            }
            // Deny all other station/standard rights even through Users/Everyone.
            descriptor.DiscretionaryAcl.InsertAce(0, new CommonAce(AceFlags.None, AceQualifier.AccessDenied, 0x000d037d, sid, false, null));
            descriptor.DiscretionaryAcl.InsertAce(1, new CommonAce(AceFlags.None, AceQualifier.AccessAllowed, 0x00020002, sid, false, null));
            if (descriptor.BinaryLength > 8192) throw new InvalidOperationException("Station ACL proposal exceeds recovery bound");
            byte[] result = new byte[descriptor.BinaryLength]; descriptor.GetBinaryForm(result, 0); return result;
        }
        public static HostedDesktop Prepare(string owner, string candidate, string recoveryDirectory) {
            var value = new HostedDesktop();
            try {
                if (WindowsIdentity.GetCurrent().User.Value != owner || owner == candidate) throw new InvalidOperationException("Distinct actual owner required");
                value.station = OpenWindowStationW("WinSta0", false, 0x0006000a);
                if (value.station == IntPtr.Zero) throw new Win32Exception(Marshal.GetLastWin32Error(), "Existing WinSta0 unavailable");
                if (ObjectName(GetProcessWindowStation()) != "WinSta0") throw new InvalidOperationException("Original broker must already belong to WinSta0");
                UserObjectFlags flags; uint flagBytes;
                if (!GetUserObjectInformationW(value.station, 1, out flags, (uint)Marshal.SizeOf<UserObjectFlags>(), out flagBytes) || (flags.flags & 1) == 0)
                    throw new InvalidOperationException("Existing station is not interactive");
                value.before = ReadAcl(value.station);
                byte[] proposal = StationProposal(value.before, candidate);
                // Owner-only storage was prepared by Setup.Directory before this call.
                using (var file = new System.IO.FileStream(System.IO.Path.Combine(recoveryDirectory, "station-dacl-before.bin"), System.IO.FileMode.CreateNew)) { file.Write(value.before, 0, value.before.Length); file.Flush(true); }
                using (var file = new System.IO.FileStream(System.IO.Path.Combine(recoveryDirectory, "station-dacl-proposed.bin"), System.IO.FileMode.CreateNew)) { file.Write(proposal, 0, proposal.Length); file.Flush(true); }
                value.applied = proposal; value.changed = true;
                WriteAcl(value.station, proposal);
                if (!System.Linq.Enumerable.SequenceEqual(ReadAcl(value.station), proposal)) throw new InvalidOperationException("Station DACL postimage differs");
                value.Name = "PortcoveHosted_" + Guid.NewGuid().ToString("N");
                IntPtr previous = GetProcessWindowStation();
                if (previous == IntPtr.Zero || !SetProcessWindowStation(value.station)) throw new Win32Exception(Marshal.GetLastWin32Error(), "Select hosted station");
                try {
                    IntPtr existing = OpenDesktopW(value.Name, 0, false, 0x20081);
                    if (existing != IntPtr.Zero) { CloseDesktop(existing); throw new InvalidOperationException("Existing desktop refused"); }
                    if (Marshal.GetLastWin32Error() != 2) throw new Win32Exception(Marshal.GetLastWin32Error(), "Desktop absence unavailable");
                    var acl = new RawSecurityDescriptor("O:" + owner + "D:P(A;;GA;;;" + owner + ")(A;;GA;;;SY)(A;;0x000200c7;;;" + candidate + ")");
                    byte[] bytes = new byte[acl.BinaryLength]; acl.GetBinaryForm(bytes, 0);
                    var pinned = GCHandle.Alloc(bytes, GCHandleType.Pinned);
                    try {
                        var attributes = new Attributes {length=Marshal.SizeOf<Attributes>(), descriptor=pinned.AddrOfPinnedObject()};
                        value.desktop = CreateDesktopW(value.Name, IntPtr.Zero, IntPtr.Zero, 0, 0x000f01ff, ref attributes);
                        if (value.desktop == IntPtr.Zero) throw new Win32Exception(Marshal.GetLastWin32Error(), "Create non-input private desktop");
                    } finally { pinned.Free(); }
                    if (ObjectName(value.desktop) != value.Name) throw new InvalidOperationException("Private desktop name differs");
                    if (!DesktopMatches(ReadAcl(value.desktop), bytes, owner)) throw new InvalidOperationException("Private desktop DACL differs");
                    value.AssertNonInput();
                } finally { if (!SetProcessWindowStation(previous)) throw new Win32Exception(Marshal.GetLastWin32Error(), "Restore station"); }
                return value;
            } catch (Exception failure) {
                try { value.Dispose(); } catch (Exception recovery) { throw new AggregateException("Hosted desktop preparation and recovery refused", failure, recovery); }
                throw;
            }
        }
        [DllImport("user32", CharSet=CharSet.Unicode, SetLastError=true)] static extern bool GetUserObjectInformationW(IntPtr handle, int index, StringBuilder value, uint length, out uint needed);
        [StructLayout(LayoutKind.Sequential)] struct UserObjectFlags { public int inherit, reserved; public uint flags; }
        [DllImport("user32", CharSet=CharSet.Unicode, SetLastError=true)] static extern bool GetUserObjectInformationW(IntPtr handle, int index, out UserObjectFlags value, uint length, out uint needed);
        static string ObjectName(IntPtr handle) {
            var name = new StringBuilder(256); uint needed;
            if (!GetUserObjectInformationW(handle, 2, name, 512, out needed)) throw new Win32Exception(Marshal.GetLastWin32Error(), "Read bounded station name");
            return name.ToString();
        }
        static bool DesktopMatches(byte[] actual, byte[] expected, string owner) {
            var observed = new RawSecurityDescriptor(actual, 0); var proposal = new RawSecurityDescriptor(expected, 0);
            if (observed.Owner.Value != owner || (observed.ControlFlags & ControlFlags.DiscretionaryAclProtected) == 0) return false;
            byte[] a = new byte[observed.DiscretionaryAcl.BinaryLength], b = new byte[proposal.DiscretionaryAcl.BinaryLength];
            observed.DiscretionaryAcl.GetBinaryForm(a, 0); proposal.DiscretionaryAcl.GetBinaryForm(b, 0);
            return System.Linq.Enumerable.SequenceEqual(a, b);
        }
        public void AssertNonInput() {
            int input; uint needed;
            if (!GetUserObjectInformationW(desktop, 6, out input, 4, out needed) || input != 0) throw new InvalidOperationException("Private desktop must remain non-input");
        }
        public void Dispose() {
            try {
                if (changed) {
                    byte[] current = ReadAcl(station);
                    if (!System.Linq.Enumerable.SequenceEqual(current, before)) {
                        if (!System.Linq.Enumerable.SequenceEqual(current, applied)) throw new InvalidOperationException("Station ACL changed concurrently; retain recovery, no overwrite");
                        WriteAcl(station, before);
                    }
                    if (!System.Linq.Enumerable.SequenceEqual(ReadAcl(station), before)) throw new InvalidOperationException("Station ACL restoration unavailable");
                    changed = false; Restored = true;
                }
            } finally {
                if (desktop != IntPtr.Zero) { CloseDesktop(desktop); desktop = IntPtr.Zero; }
                if (station != IntPtr.Zero) { CloseWindowStation(station); station = IntPtr.Zero; }
            }
        }
    }
