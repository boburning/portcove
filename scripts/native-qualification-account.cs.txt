    public sealed class AccountProcess : IDisposable {
        IntPtr job, process, station, desktop;
        readonly uint pid;
        readonly long created;
        readonly string executable;
        public Task<string> Output { get; private set; }
        public Task<string> ErrorOutput { get; private set; }
        bool stopAttempted;
        internal AccountProcess(IntPtr jobHandle, IntPtr processHandle, uint processId, IntPtr stationHandle, IntPtr desktopHandle, string executablePath, IntPtr outputRead, IntPtr errorRead) {
            job = jobHandle; process = processHandle; pid = processId;
            station = stationHandle; desktop = desktopHandle;
            executable = executablePath;
            long ignored1, ignored2, ignored3;
            if (!Native.GetProcessTimes(process, out created, out ignored1, out ignored2, out ignored3))
                throw Native.Error("GetProcessTimes");
            Output = Capture(outputRead); ErrorOutput = Capture(errorRead);
        }
        static Task<string> Capture(IntPtr pipe) {
            // Anonymous CreatePipe handles are synchronous. Start the entire
            // read on a worker, rather than blocking before the suspended
            // child can resume or its writer can produce the first byte.
            return Task.Run(async () => {
            using (var stream = new FileStream(new SafeFileHandle(pipe, true), FileAccess.Read, 4096, false))
            using (var retained = new MemoryStream()) {
                byte[] buffer = new byte[4096];
                while (true) {
                    int count = await stream.ReadAsync(buffer, 0, buffer.Length).ConfigureAwait(false);
                    if (count == 0) break;
                    if (retained.Length + count > 4 * 1024 * 1024) throw new IOException("Native output exceeded fixed four-MiB bound; no successful capture");
                    retained.Write(buffer, 0, count);
                }
                return new UTF8Encoding(false, true).GetString(retained.ToArray());
            }
            });
        }
        public uint Pid { get { return pid; } }
        public int Id { get { return checked((int)pid); } }
        public long CreatedFileTime { get { return created; } }
        public DateTime StartTime { get { return DateTime.FromFileTimeUtc(created); } }
        public IntPtr Handle { get { return process; } }
        public ProcessStartInfo StartInfo { get { return new ProcessStartInfo(executable); } }
        public bool HasExited { get { return WaitForExit(0); } }
        public int ExitCode {
            get {
                if (!HasExited) throw new InvalidOperationException("Root process has not exited");
                uint exit; if (!Native.GetExitCodeProcess(process, out exit)) throw Native.Error("GetExitCodeProcess");
                return unchecked((int)exit);
            }
        }
        public string Path {
            get {
                uint length = 32768; var result = new StringBuilder((int)length);
                if (!Native.QueryFullProcessImageNameW(process, 0, result, ref length)) throw Native.Error("QueryFullProcessImageNameW");
                return result.ToString();
            }
        }
        public void Refresh() { /* Each property reads the retained native handle. */ }
        public TeardownObservation LastTeardown { get; private set; }
        IntPtr FindRootWindow() {
            IntPtr found = IntPtr.Zero;
            Native.DesktopWindowCallback callback = delegate(IntPtr window, IntPtr unused) {
                uint owner; Native.GetWindowThreadProcessId(window, out owner);
                if (owner == pid && Native.IsWindowVisible(window)) { found = window; return false; }
                return true;
            };
            Native.EnumDesktopWindows(desktop, callback, IntPtr.Zero);
            GC.KeepAlive(callback);
            return found;
        }
        public IntPtr MainWindowHandle { get { return FindRootWindow(); } }
        public string MainWindowTitle {
            get {
                IntPtr window = FindRootWindow();
                if (window == IntPtr.Zero) return String.Empty;
                var title = new StringBuilder(1024); Native.GetWindowTextW(window, title, title.Capacity);
                return title.ToString();
            }
        }
        public bool Responding {
            get {
                IntPtr window = FindRootWindow(); if (window == IntPtr.Zero) return false;
                UIntPtr result;
                return Native.SendMessageTimeoutW(window, 0, UIntPtr.Zero, IntPtr.Zero, 3, 1000, out result) != IntPtr.Zero;
            }
        }
        public bool CloseMainWindow() {
            IntPtr window = FindRootWindow();
            return window != IntPtr.Zero && Native.PostMessageW(window, 0x10, UIntPtr.Zero, IntPtr.Zero);
        }
        public void Kill(bool entireProcessTree) {
            if (!entireProcessTree) throw new ArgumentException("Native qualification always terminates the whole owned job");
            StopAndObserve(5000);
        }
        public bool WaitForExit(int milliseconds) {
            if (milliseconds < 0 || milliseconds > 600000) throw new ArgumentOutOfRangeException("milliseconds");
            uint outcome = Native.WaitForSingleObject(process, (uint)milliseconds);
            if (outcome == 0xffffffff) throw Native.Error("WaitForSingleObject");
            return outcome == 0;
        }
        // The job stays open until positive empty readback. Unknown teardown never permits credential release.
        public TeardownObservation StopAndObserve(int milliseconds) {
            if (milliseconds < 1 || milliseconds > 30000) throw new ArgumentOutOfRangeException("milliseconds");
            if (stopAttempted) throw new InvalidOperationException("Termination attempt is one-shot; use readback for ambiguous outcomes");
            stopAttempted = true;
            if (!Native.TerminateJobObject(job, 125)) throw Native.Error("TerminateJobObject");
            return ReadTeardown(milliseconds);
        }
        public TeardownObservation ReadTeardown(int milliseconds) {
            if (milliseconds < 1 || milliseconds > 30000) throw new ArgumentOutOfRangeException("milliseconds");
            Stopwatch clock = Stopwatch.StartNew();
            uint[] remaining;
            do {
                remaining = Native.JobPids(job);
                if (remaining.Length == 0) break;
                Thread.Sleep(25);
            } while (clock.ElapsedMilliseconds < milliseconds);
            bool exited = Native.WaitForSingleObject(process, 0) == 0;
            uint exit;
            if (!Native.GetExitCodeProcess(process, out exit)) throw Native.Error("GetExitCodeProcess");
            LastTeardown = new TeardownObservation {
                RootPid = pid, RootCreatedFileTime = created, ExitCode = exit,
                RootExited = exited, JobEmpty = remaining.Length == 0 && exited, RemainingPids = remaining,
                ObservedAt = DateTime.UtcNow.ToString("o")
            };
            return LastTeardown;
        }
        public void Dispose() {
            // Kill-on-close is a last resort, never a positive teardown assertion.
            if (job != IntPtr.Zero) { Native.CloseHandle(job); job = IntPtr.Zero; }
            if (process != IntPtr.Zero) { Native.CloseHandle(process); process = IntPtr.Zero; }
            if (desktop != IntPtr.Zero) { Native.CloseDesktop(desktop); desktop = IntPtr.Zero; }
            if (station != IntPtr.Zero) { Native.CloseWindowStation(station); station = IntPtr.Zero; }
        }
    }

    public static class AccountLauncher {
        public const string CredentialTarget = "Portcove/native-qualification/local-account/v1";
        // Called only by the configured broker after its setup/grant guard.
        // This never reads a GitHub App, semantic, signing or publication key.
        public static SecureString ReadAccountCredential(AccountConfiguration config) {
            if (config == null || !new SecurityIdentifier(config.BrokerSid).Equals(WindowsIdentity.GetCurrent().User)) throw new ArgumentException("Wrong credential owner");
            IntPtr record;
            if (!Native.CredReadW(CredentialTarget, 1, 0, out record)) throw Native.Error("fixed local account credential unavailable");
            Native.Credential credential = new Native.Credential();
            try {
                credential = Marshal.PtrToStructure<Native.Credential>(record);
                string user = Marshal.PtrToStringUni(credential.user);
                if (!String.Equals(user, config.UserName, StringComparison.OrdinalIgnoreCase) && !String.Equals(user, Environment.MachineName + "\\" + config.UserName, StringComparison.OrdinalIgnoreCase)) throw new InvalidOperationException("Stored local credential account differs");
                if (credential.type != 1 || credential.blobSize == 0 || credential.blobSize > 512 || credential.blobSize % 2 != 0 || credential.blob == IntPtr.Zero) throw new InvalidOperationException("Local credential format exceeds fixed boundary");
                var secret = new SecureString();
                for (int index = 0; index < credential.blobSize; index += 2) {
                    char value = (char)Marshal.ReadInt16(credential.blob, index);
                    if (value == '\0') { secret.Dispose(); throw new InvalidOperationException("Null local credential character refused"); }
                    secret.AppendChar(value);
                }
                secret.MakeReadOnly(); return secret;
            } finally {
                if (credential.blob != IntPtr.Zero && credential.blobSize <= 512)
                    for (int index = 0; index < credential.blobSize; index++) Marshal.WriteByte(credential.blob, index, 0);
                Native.CredFree(record);
            }
        }
        static readonly HashSet<string> EnvironmentNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase) {
            "SystemRoot", "WINDIR", "PATH", "USERPROFILE", "HOMEDRIVE", "HOMEPATH", "APPDATA", "LOCALAPPDATA", "TEMP", "TMP"
        };
        public static string Quote(string value) {
            if (value == null || value.IndexOf('\0') >= 0) throw new ArgumentException("Invalid argument");
            if (value.Length > 0 && !value.Any(item => item == ' ' || item == '\t' || item == '"')) return value;
            StringBuilder result = new StringBuilder("\""); int slashes = 0;
            foreach (char item in value) {
                if (item == '\\') { slashes++; continue; }
                result.Append('\\', item == '"' ? slashes * 2 + 1 : slashes);
                slashes = 0; result.Append(item);
            }
            return result.Append('\\', slashes * 2).Append('"').ToString();
        }
        public static string NsisCommandLine(string executable, string[] arguments, string workRoot) {
            if (arguments == null || arguments.Length < 1 || arguments.Length > 3) throw new ArgumentException("Fixed NSIS arguments required");
            string tail = arguments[arguments.Length - 1];
            bool destination = tail.StartsWith("/D=", StringComparison.Ordinal) || tail.StartsWith("_?=", StringComparison.Ordinal);
            var flags = destination ? arguments.Take(arguments.Length - 1) : arguments;
            if (flags.Any(flag => flag != "/S" && flag != "/P" && flag != "/UPDATE") || flags.Distinct().Count() != flags.Count()) throw new ArgumentException("Unexpected NSIS switch");
            string command = Quote(executable) + " " + String.Join(" ", flags);
            if (destination) {
                string target = tail.Substring(3);
                if (target.Any(item => item < 32 || item == '"') || !System.IO.Path.IsPathFullyQualified(target) || target.StartsWith("\\\\") || !System.IO.Path.GetFullPath(target).StartsWith(System.IO.Path.GetFullPath(workRoot).TrimEnd('\\') + "\\", StringComparison.OrdinalIgnoreCase)) throw new ArgumentException("NSIS destination must stay in qualification work");
                // NSIS explicitly requires /D= and _?= last and unquoted even
                // when their absolute path contains spaces. No shell is used.
                command += " " + tail;
            }
            if (command.Length >= 1024) throw new ArgumentException("NSIS command exceeds fixed logon limit");
            return command;
        }
        public static void Validate(AccountConfiguration config, string executable, string[] arguments, string directory, IDictionary<string,string> environment) {
            if (config == null || environment == null || arguments == null) throw new ArgumentException("Missing configuration");
            if (config.UserName == null || !System.Text.RegularExpressions.Regex.IsMatch(config.UserName, "^[A-Za-z][A-Za-z0-9_-]{1,31}$")) throw new ArgumentException("Fixed local account required");
            var account = new SecurityIdentifier(config.AccountSid);
            var broker = new SecurityIdentifier(config.BrokerSid);
            if (account.Equals(broker) || account.IsWellKnown(WellKnownSidType.LocalSystemSid)) throw new ArgumentException("Distinct non-system account required");
            if (!broker.Equals(WindowsIdentity.GetCurrent().User)) throw new ArgumentException("Wrong broker owner");
            if (String.IsNullOrEmpty(config.PrivateDesktop) || !System.Text.RegularExpressions.Regex.IsMatch(config.PrivateDesktop, "^PortcoveNative_[0-9a-f]{32}\\\\Qualification$")) throw new ArgumentException("Private desktop required");
            foreach (string path in new [] { executable, directory, config.ProfileRoot, config.WorkRoot }) {
                if (String.IsNullOrEmpty(path) || !Path.IsPathFullyQualified(path) || path.StartsWith("\\\\") || path.IndexOf('\0') >= 0) throw new ArgumentException("Local absolute path required");
                for (string cursor = Path.GetFullPath(path); !String.IsNullOrEmpty(cursor); cursor = Path.GetDirectoryName(cursor))
                    if ((File.Exists(cursor) || Directory.Exists(cursor)) && (File.GetAttributes(cursor) & FileAttributes.ReparsePoint) != 0) throw new ArgumentException("Reparse ancestry refused");
            }
            string work = Path.GetFullPath(config.WorkRoot).TrimEnd('\\') + "\\";
            if (!Path.GetFullPath(directory).StartsWith(work, StringComparison.OrdinalIgnoreCase)) throw new ArgumentException("Working directory outside qualification root");
            string command = String.Join(" ", new [] { Quote(executable) }.Concat(arguments.Select(Quote)));
            if (command.Length >= 1024) throw new ArgumentException("CreateProcessWithLogonW command exceeds fixed limit");
            if (environment.Count != EnvironmentNames.Count || environment.Keys.Any(key => !EnvironmentNames.Contains(key))) throw new ArgumentException("Exact credential-free environment required");
            foreach (var item in environment) if (item.Value == null || item.Value.IndexOf('\0') >= 0) throw new ArgumentException("Invalid environment");
            if (!String.Equals(environment["USERPROFILE"], config.ProfileRoot, StringComparison.OrdinalIgnoreCase)) throw new ArgumentException("Profile mismatch");
            string profile = Path.GetFullPath(config.ProfileRoot);
            if (!String.Equals(environment["HOMEDRIVE"], Path.GetPathRoot(profile).TrimEnd('\\'), StringComparison.OrdinalIgnoreCase) || !String.Equals(environment["HOMEPATH"], profile.Substring(Path.GetPathRoot(profile).Length - 1), StringComparison.OrdinalIgnoreCase)) throw new ArgumentException("Home path mismatch");
            string system = Environment.GetFolderPath(Environment.SpecialFolder.Windows);
            if (!String.Equals(environment["SystemRoot"], system, StringComparison.OrdinalIgnoreCase) || !String.Equals(environment["WINDIR"], system, StringComparison.OrdinalIgnoreCase) || !String.Equals(environment["PATH"], Path.Combine(system, "System32"), StringComparison.OrdinalIgnoreCase)) throw new ArgumentException("Untrusted executable search path");
            foreach (string name in new [] { "TEMP", "TMP", "APPDATA", "LOCALAPPDATA" })
                if (!Path.GetFullPath(environment[name]).StartsWith(work, StringComparison.OrdinalIgnoreCase)) throw new ArgumentException("Writable environment path outside qualification root");
        }
        public static AccountProcess Start(AccountConfiguration config, SecureString password, string executable, string[] arguments, string directory, IDictionary<string,string> environment) {
            return StartCore(config, password, executable, arguments, directory, environment, false);
        }
        public static AccountProcess StartNsis(AccountConfiguration config, SecureString password, string executable, string[] arguments, string directory, IDictionary<string,string> environment) {
            return StartCore(config, password, executable, arguments, directory, environment, true);
        }
        static AccountProcess StartCore(AccountConfiguration config, SecureString password, string executable, string[] arguments, string directory, IDictionary<string,string> environment, bool nsis) {
            Validate(config, executable, arguments, directory, environment);
            string commandText = nsis ? NsisCommandLine(executable, arguments, config.WorkRoot) : String.Join(" ", new [] { Quote(executable) }.Concat(arguments.Select(Quote)));
            if (password == null || password.Length < 1) throw new ArgumentException("Fixed owner-held account credential unavailable");
            IntPtr job = IntPtr.Zero, secret = IntPtr.Zero, block = IntPtr.Zero;
            IntPtr station = IntPtr.Zero, desktop = IntPtr.Zero;
            IntPtr outputRead = IntPtr.Zero, outputWrite = IntPtr.Zero;
            IntPtr errorRead = IntPtr.Zero, errorWrite = IntPtr.Zero, input = IntPtr.Zero;
            Native.ProcessInformation info = new Native.ProcessInformation(); bool transferred = false;
            try {
                // This fixed helper must run in its own noninteractive owner process:
                // SetProcessWindowStation is process-wide and never targets WinSta0.
                Native.CreatePrivateDesktop(config, out station, out desktop);
                job = Native.CreateJobObjectW(IntPtr.Zero, null);
                if (job == IntPtr.Zero) throw Native.Error("CreateJobObjectW");
                var limits = new Native.ExtendedLimitInformation();
                limits.basic.limitFlags = 0x2000 | 0x8; // Kill-on-close, total active processes <=64. No breakaway flags.
                limits.basic.activeProcessLimit = 64;
                if (!Native.SetInformationJobObject(job, 9, ref limits, (uint)Marshal.SizeOf(limits))) throw Native.Error("SetInformationJobObject");
                var pipeAttributes = new Native.SecurityAttributes { length = (uint)Marshal.SizeOf(typeof(Native.SecurityAttributes)), inherit = 1 };
                if (!Native.CreatePipe(out outputRead, out outputWrite, ref pipeAttributes, 0) || !Native.SetHandleInformation(outputRead, 1, 0)) throw Native.Error("stdout pipe");
                if (!Native.CreatePipe(out errorRead, out errorWrite, ref pipeAttributes, 0) || !Native.SetHandleInformation(errorRead, 1, 0)) throw Native.Error("stderr pipe");
                input = Native.CreateFileW("NUL", 0x80000000, 3, ref pipeAttributes, 3, 0, IntPtr.Zero);
                if (input == new IntPtr(-1)) throw Native.Error("credential-free null stdin");
                secret = Marshal.SecureStringToGlobalAllocUnicode(password);
                string environmentBlock = String.Join("\0", environment.OrderBy(item => item.Key, StringComparer.OrdinalIgnoreCase).Select(item => item.Key + "=" + item.Value)) + "\0\0";
                block = Marshal.StringToHGlobalUni(environmentBlock);
                var startup = new Native.StartupInfo(); startup.cb = (uint)Marshal.SizeOf(startup);
                startup.desktop = config.PrivateDesktop; startup.flags = 1 | 0x100; startup.showWindow = 0;
                startup.input = input; startup.output = outputWrite; startup.error = errorWrite;
                var command = new StringBuilder(commandText);
                if (!Native.CreateProcessWithLogonW(config.UserName, ".", secret, 1, executable, command, 0x4 | 0x400, block, directory, ref startup, out info)) throw Native.Error("CreateProcessWithLogonW");
                // No candidate thread has run. Assign job before token inspection/resume.
                if (!Native.AssignProcessToJobObject(job, info.process)) throw Native.Error("AssignProcessToJobObject");
                IntPtr token;
                if (!Native.OpenProcessToken(info.process, 8, out token)) throw Native.Error("OpenProcessToken");
                try {
                    using (var identity = new WindowsIdentity(token)) {
                        if (!identity.User.Equals(new SecurityIdentifier(config.AccountSid))) throw new InvalidOperationException("Launched account SID mismatch");
                        if (identity.Groups.Any(group => group.Equals(new SecurityIdentifier(WellKnownSidType.BuiltinAdministratorsSid, null)))) throw new InvalidOperationException("Administrator account refused, including filtered administrator tokens");
                    }
                    uint profileLength = 32768; var loadedProfile = new StringBuilder((int)profileLength);
                    if (!Native.GetUserProfileDirectoryW(token, loadedProfile, ref profileLength)) throw Native.Error("GetUserProfileDirectoryW");
                    if (!String.Equals(Path.GetFullPath(loadedProfile.ToString()).TrimEnd('\\'), Path.GetFullPath(config.ProfileRoot).TrimEnd('\\'), StringComparison.OrdinalIgnoreCase)) throw new InvalidOperationException("Actual loaded account profile mismatch");
                    uint childSession, brokerSession;
                    if (!Native.ProcessIdToSessionId(info.pid, out childSession) || !Native.ProcessIdToSessionId(Native.GetCurrentProcessId(), out brokerSession) || childSession != brokerSession) throw new InvalidOperationException("Native GUI session mismatch");
                } finally { Native.CloseHandle(token); }
                var session = new AccountProcess(job, info.process, info.pid, station, desktop, executable, outputRead, errorRead);
                outputRead = IntPtr.Zero; errorRead = IntPtr.Zero;
                if (!String.Equals(System.IO.Path.GetFullPath(session.Path), System.IO.Path.GetFullPath(executable), StringComparison.OrdinalIgnoreCase)) throw new InvalidOperationException("Suspended process image differs from exact requested path");
                if (Native.ResumeThread(info.thread) == 0xffffffff) throw Native.Error("ResumeThread");
                transferred = true; return session;
            } finally {
                if (secret != IntPtr.Zero) Marshal.ZeroFreeGlobalAllocUnicode(secret);
                if (block != IntPtr.Zero) Marshal.FreeHGlobal(block);
                if (info.thread != IntPtr.Zero) Native.CloseHandle(info.thread);
                foreach (IntPtr handle in new [] { outputRead, errorRead, outputWrite, errorWrite, input })
                    if (handle != IntPtr.Zero && handle != new IntPtr(-1)) Native.CloseHandle(handle);
                if (!transferred) {
                    if (info.process != IntPtr.Zero) { Native.TerminateProcess(info.process, 125); Native.WaitForSingleObject(info.process, 30000); Native.CloseHandle(info.process); }
                    if (job != IntPtr.Zero) Native.CloseHandle(job);
                    if (desktop != IntPtr.Zero) Native.CloseDesktop(desktop);
                    if (station != IntPtr.Zero) Native.CloseWindowStation(station);
                }
            }
        }
    }

        internal static void CreatePrivateDesktop(AccountConfiguration config, out IntPtr station, out IntPtr desktop) {
            station = IntPtr.Zero; desktop = IntPtr.Zero;
            IntPtr descriptor = IntPtr.Zero, previous = GetProcessWindowStation();
            if (previous == IntPtr.Zero) throw Error("GetProcessWindowStation");
            uint size;
            // Only the broker and SYSTEM control the object DACL. Candidate
            // receives specific GUI rights and READ_CONTROL, no WRITE_DAC/OWNER.
            string prefix = "O:" + config.BrokerSid + "D:P(A;;GA;;;" + config.BrokerSid + ")(A;;GA;;;SY)";
            try {
                if (!ConvertStringSecurityDescriptorToSecurityDescriptorW(prefix + "(A;;0x0002037f;;;" + config.AccountSid + ")", 1, out descriptor, out size)) throw Error("window station security descriptor");
                var attributes = new SecurityAttributes { length = (uint)Marshal.SizeOf(typeof(SecurityAttributes)), descriptor = descriptor, inherit = 0 };
                station = CreateWindowStationW(config.PrivateDesktop.Split('\\')[0], 0, 0x000f037f, ref attributes);
                int error = Marshal.GetLastWin32Error();
                if (station == IntPtr.Zero || error == 183) throw new Win32Exception(error, "private window station must be new");
                LocalFree(descriptor); descriptor = IntPtr.Zero;
                if (!SetProcessWindowStation(station)) throw Error("SetProcessWindowStation(private)");
                try {
                    if (!ConvertStringSecurityDescriptorToSecurityDescriptorW(prefix + "(A;;0x000201ff;;;" + config.AccountSid + ")", 1, out descriptor, out size)) throw Error("desktop security descriptor");
                    attributes.descriptor = descriptor;
                    desktop = CreateDesktopW("Qualification", IntPtr.Zero, IntPtr.Zero, 0, 0x000f01ff, ref attributes);
                    if (desktop == IntPtr.Zero) throw Error("CreateDesktopW");
                } finally {
                    if (!SetProcessWindowStation(previous)) throw Error("SetProcessWindowStation(restore)");
                }
            } finally { if (descriptor != IntPtr.Zero) LocalFree(descriptor); }
        }
        internal static uint[] JobPids(IntPtr job) {
            int size = 8 + 64 * IntPtr.Size; IntPtr buffer = Marshal.AllocHGlobal(size);
            try {
                uint returned;
                if (!QueryInformationJobObject(job, 3, buffer, (uint)size, out returned)) throw Error("QueryInformationJobObject");
                int assigned = Marshal.ReadInt32(buffer, 0), count = Marshal.ReadInt32(buffer, 4);
                if (assigned != count || count < 0 || count > 64) throw new InvalidOperationException("Incomplete job process inventory");
                uint[] pids = new uint[count];
                for (int index = 0; index < count; index++) pids[index] = checked((uint)Marshal.ReadIntPtr(buffer, 8 + index * IntPtr.Size).ToInt64());
                return pids;
            } finally { Marshal.FreeHGlobal(buffer); }
        }
