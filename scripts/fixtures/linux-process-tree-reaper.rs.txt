//! Job-owned Linux adopting supervisor. No dependency or persistent host service.
use std::env;
use std::fs::{self, OpenOptions};
use std::io::{self, Write};
use std::os::unix::process::CommandExt;
use std::path::Path;
use std::process::{Command, ExitCode, Stdio};
use std::time::{Duration, Instant};

#[repr(C)]
struct PollFd {
    fd: i32,
    events: i16,
    revents: i16,
}

unsafe extern "C" {
    fn prctl(option: i32, ...) -> i32;
    fn setpgid(pid: i32, group: i32) -> i32;
    fn kill(pid: i32, signal: i32) -> i32;
    fn waitpid(pid: i32, status: *mut i32, options: i32) -> i32;
    fn poll(fds: *mut PollFd, count: usize, timeout: i32) -> i32;
}

fn last_error() -> io::Error {
    io::Error::last_os_error()
}

fn atomic_file(path: &Path, contents: &str) -> io::Result<()> {
    let pending = path.with_extension(format!("pending-{}", std::process::id()));
    let mut file = OpenOptions::new().write(true).create_new(true).open(&pending)?;
    file.write_all(contents.as_bytes())?;
    file.sync_all()?;
    // Atomic no-clobber publication also refuses a late receipt or symlink.
    fs::hard_link(&pending, path)?;
    fs::remove_file(pending)
}

fn stat(pid: i32) -> io::Result<Vec<String>> {
    let value = fs::read_to_string(format!("/proc/{pid}/stat"))?;
    let closing = value.rfind(')').ok_or_else(|| io::Error::other("invalid process stat"))?;
    let fields: Vec<_> = value[closing + 2..].split_whitespace().map(str::to_owned).collect();
    if fields.len() < 20 {
        return Err(io::Error::other("incomplete process stat"));
    }
    Ok(fields)
}

fn pipe_lost() -> io::Result<bool> {
    let mut descriptor = PollFd { fd: 0, events: 1, revents: 0 };
    let result = unsafe { poll(&mut descriptor, 1, 10) };
    if result < 0 {
        let error = last_error();
        return if error.raw_os_error() == Some(4) { Ok(false) } else { Err(error) };
    }
    if descriptor.revents & (8 | 32) != 0 {
        return Err(io::Error::other("wrapper control pipe failed"));
    }
    if descriptor.revents & 1 != 0 {
        return Err(io::Error::other("unexpected wrapper control pipe input"));
    }
    Ok(descriptor.revents & 16 != 0)
}

fn close_group(group: i32) -> io::Result<()> {
    // The unreaped direct Node child pins this group identity until this signal.
    if unsafe { kill(-group, 9) } < 0 {
        let error = last_error();
        if error.raw_os_error() != Some(3) { return Err(error); }
    }
    Ok(())
}

fn drain_and_prove_absence(group: i32) -> io::Result<()> {
    let deadline = Instant::now() + Duration::from_millis(5_000);
    loop {
        let mut status = 0;
        let mut no_children = false;
        loop {
            if Instant::now() >= deadline {
                return Err(io::Error::other("owned child reaping exceeded cleanup deadline"));
            }
            let waited = unsafe { waitpid(-1, &mut status, 1) };
            if waited > 0 { continue; }
            if waited == 0 { break; }
            let error = last_error();
            match error.raw_os_error() {
                Some(4) => continue,
                Some(10) => { no_children = true; break; }
                _ => return Err(error),
            }
        }
        let absent = if unsafe { kill(-group, 0) } < 0 {
            let error = last_error();
            if error.raw_os_error() != Some(3) { return Err(error); }
            true
        } else { false };
        if no_children && absent { return Ok(()); }
        if Instant::now() >= deadline {
            return Err(io::Error::other("owned children or anchored group remain at cleanup deadline"));
        }
        std::thread::sleep(Duration::from_millis(10));
    }
}

fn run(args: &[String], group: &mut Option<i32>) -> io::Result<()> {
    // Move the adopter outside the payload group while retaining a common session.
    if unsafe { setpgid(0, 0) } < 0 { return Err(last_error()); }
    if unsafe { prctl(36, 1usize, 0usize, 0usize, 0usize) } < 0 { return Err(last_error()); }
    if !fs::read_link("/proc/self/fd/0")?.to_string_lossy().starts_with("pipe:[") {
        return Err(io::Error::other("wrapper control input is not an owned pipe"));
    }
    if pipe_lost()? { return Err(io::Error::other("wrapper lost before supervisor spawn")); }

    let mut command = Command::new(&args[4]);
    command.arg(&args[5]).arg("--linux-managed")
        .args(&args[..3]).args(&args[6..]).stdin(Stdio::piped());
    unsafe {
        command.pre_exec(|| {
            if setpgid(0, 0) < 0 { return Err(last_error()); }
            Ok(())
        });
    }
    let child = command.spawn()?;
    let pid = i32::try_from(child.id()).map_err(|_| io::Error::other("invalid Node supervisor PID"))?;
    *group = Some(pid);
    // Do not try_wait/wait here: retain the Node PID until anchored cleanup starts.
    let fields = stat(pid)?;
    if fields[1] != std::process::id().to_string() || fields[2] != pid.to_string() {
        return Err(io::Error::other("Node supervisor parent/group setup mismatch"));
    }
    let boot = fs::read_to_string("/proc/sys/kernel/random/boot_id")?;
    let boot = boot.trim();
    if !boot.bytes().all(|value| value.is_ascii_hexdigit() || value == b'-') || boot.len() != 36
        || !fields[19].bytes().all(|value| value.is_ascii_digit()) {
        return Err(io::Error::other("invalid supervisor birth identity"));
    }
    atomic_file(Path::new(&args[3]), &format!(
        "{{\"pid\":{pid},\"identity\":\"linux:{boot}:{pid}:{}\"}}\n", fields[19]
    ))?;
    loop {
        if pipe_lost()? || stat(pid)?[0] == "Z" { break; }
    }
    Ok(())
}

fn main() -> ExitCode {
    let args: Vec<_> = env::args().skip(1).collect();
    if args.len() < 7 {
        eprintln!("usage: linux-process-tree-reaper <gate> <status> <receipt> <registration> <node> <supervisor> <command> [arguments...]");
        return ExitCode::from(2);
    }
    let mut group = None;
    let initial = run(&args, &mut group);
    // Even setup/pipe/publication failure must attempt closure of an admitted child.
    let cleanup = match group {
        Some(pid) => close_group(pid).and_then(|()| drain_and_prove_absence(pid)),
        None => Ok(()),
    };
    let outcome = match initial.and(cleanup) {
        Ok(()) => "quiescent",
        Err(error) => {
            eprintln!("Linux Heavy Rust reaper failed closed: {error}");
            "failed:linux-reaper"
        }
    };
    if let Err(error) = atomic_file(Path::new(&args[2]), &format!("{{\"outcome\":\"{outcome}\"}}\n")) {
        eprintln!("Linux Heavy Rust cleanup receipt publication failed: {error}");
        return ExitCode::from(1);
    }
    ExitCode::from(u8::from(outcome != "quiescent"))
}
