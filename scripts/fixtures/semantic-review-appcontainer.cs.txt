  public sealed class BoundaryLocks : IDisposable {

  public static BoundaryLocks LockBoundary(string root) {
    var held=new BoundaryLocks();
    try {
      var paths=new System.Collections.Generic.List<string>();
      for(var p=new DirectoryInfo(root);p!=null;p=p.Parent)paths.Add(p.FullName);
      var pending=new System.Collections.Generic.Stack<string>();pending.Push(root);
      while(pending.Count>0)foreach(var p in Directory.EnumerateFileSystemEntries(pending.Pop())) {
        if(paths.Count>=30000)throw new InvalidOperationException("run inventory bound");
        var attrs=File.GetAttributes(p);if((attrs&FileAttributes.ReparsePoint)!=0)throw new InvalidOperationException("reparse boundary refused");
        paths.Add(p);if((attrs&FileAttributes.Directory)!=0)pending.Push(p);
      }
      foreach(var p in paths) {
        var sa=new SA {length=Marshal.SizeOf(typeof(SA))};
        bool directory=(File.GetAttributes(p)&FileAttributes.Directory)!=0;
        uint share=directory?3u:1u;
        if(p==Path.Combine(root,".portcove-semantic-review-owned"))share=0;
        IntPtr h=CreateFileW(p,0x80,share,ref sa,3,0x02200000,IntPtr.Zero);Require(h!=new IntPtr(-1),"lock owned input/ancestry");held.Handles.Add(h);
        FileInfoNative info;Require(GetFileInformationByHandle(h,out info),"file identity");
        if((info.attributes&0x400)!=0 || (!directory&&info.links!=1))throw new InvalidOperationException("reparse/hard-linked boundary refused");
      }
      return held;
    }catch{held.Dispose();throw;}
  }
  sealed class Capture : IDisposable {
    public AnonymousPipeServerStream Pipe; public volatile bool Failed,Exceeded;
    public long Bytes; public Task Drained;
    public Capture(string path,int limit) {
      Pipe=new AnonymousPipeServerStream(PipeDirection.In,HandleInheritability.Inheritable);
      var output=new FileStream(path,FileMode.CreateNew,FileAccess.Write,FileShare.Read);
      Drained=Task.Run(()=>{try{var buffer=new byte[4096];int n;while((n=Pipe.Read(buffer,0,buffer.Length))!=0){if(Bytes+n>limit){Exceeded=true;break;}output.Write(buffer,0,n);Bytes+=n;}output.Flush(true);}catch{Failed=true;}finally{output.Dispose();}});
    }
    public void Dispose(){Pipe.Dispose();}
  }

  static string Quote(string s) {
    var b=new StringBuilder("\"");int slash=0;
    foreach(char c in s){if(c=='\\'){slash++;continue;}if(c=='\"')b.Append('\\',slash*2+1);else b.Append('\\',slash);slash=0;b.Append(c);}
    b.Append('\\',slash*2);return b.Append('"').ToString();
  }
  public static Result Run(Profile profile,string executable,string[] arguments,string cwd,string input,string stdout,string stderr,string[] environment,uint timeoutMs) {
    var handles=new System.Collections.Generic.List<IntPtr>(); IntPtr attrs=IntPtr.Zero,capMem=IntPtr.Zero,handleMem=IntPtr.Zero,envMem=IntPtr.Zero,job=IntPtr.Zero;

      handleMem=Marshal.AllocHGlobal(IntPtr.Size*3);IntPtr[] inherited={hin,hout,herr};for(int i=0;i<3;i++)Marshal.WriteIntPtr(handleMem,i*IntPtr.Size,inherited[i]);
      Require(UpdateProcThreadAttribute(attrs,0,new IntPtr(0x20002),handleMem,new IntPtr(IntPtr.Size*3),IntPtr.Zero,IntPtr.Zero),"explicit handle list");
      job=CreateJobObjectW(IntPtr.Zero,IntPtr.Zero);Require(job!=IntPtr.Zero,"job create");

      output.Pipe.DisposeLocalCopyOfClientHandle();error.Pipe.DisposeLocalCopyOfClientHandle();

      Require(AssignProcessToJobObject(job,pi.process),"assign suspended process to job");assigned=true;

      try {
        IntPtr flag=Marshal.AllocHGlobal(4);int needed;
        try{Require(GetTokenInformation(token,29,flag,4,out needed),"TokenIsAppContainer");result.IsAppContainer=Marshal.ReadInt32(flag)==1;}finally{Marshal.FreeHGlobal(flag);}
        GetTokenInformation(token,31,IntPtr.Zero,0,out needed);Require(needed>0 && needed<4096,"AppContainer SID size");
        int sidBytes=needed;IntPtr sidInfo=Marshal.AllocHGlobal(sidBytes);
        try{Require(GetTokenInformation(token,31,sidInfo,sidBytes,out needed),"TokenAppContainerSid");result.ObservedSid=SidText(Marshal.ReadIntPtr(sidInfo));}finally{Marshal.FreeHGlobal(sidInfo);}
      } finally {CloseHandle(token);}
      Require(result.IsAppContainer && result.ObservedSid==profile.Sid,"actual token differs");
      Require(ResumeThread(pi.thread)!=UInt32.MaxValue,"resume");resumed=true;
      uint wait;
      do {
        wait=WaitForSingleObject(pi.process,50);Require(wait!=UInt32.MaxValue,"wait process");
        if(watch.ElapsedMilliseconds>timeoutMs || output.Exceeded || error.Exceeded || output.Failed || error.Failed){result.ForcedTermination=true;break;}
      } while(wait==258);
      if(!result.ForcedTermination)Require(GetExitCodeProcess(pi.process,out result.ExitCode),"exit code");
      Accounting count;Require(QueryInformationJobObject(job,1,out count,Marshal.SizeOf(typeof(Accounting)),IntPtr.Zero),"query remaining tree");
      if(count.active!=0)result.ForcedTermination=true;
      if(result.ForcedTermination)Require(TerminateJobObject(job,1),"terminate owned tree");
      var stop=System.Diagnostics.Stopwatch.StartNew();
      do {
        Require(QueryInformationJobObject(job,1,out count,Marshal.SizeOf(typeof(Accounting)),IntPtr.Zero),"confirm empty tree");
        if(count.active==0){result.TreeEmpty=true;break;}System.Threading.Thread.Sleep(10);
      }while(stop.ElapsedMilliseconds<10000);
      Require(result.TreeEmpty,"tree did not become empty");
      result.CaptureComplete=Task.WaitAll(new[]{output.Drained,error.Drained},10000) && !output.Failed && !error.Failed && !output.Exceeded && !error.Exceeded;
      if(!result.CaptureComplete)result.ForcedTermination=true;
      result.StdoutBytes=output.Bytes;result.StderrBytes=error.Bytes;result.ElapsedMs=watch.ElapsedMilliseconds;return result;
    } finally {
      if(pi.process!=IntPtr.Zero){if(assigned)TerminateJobObject(job,1);else if(!resumed)TerminateProcess(pi.process,1);WaitForSingleObject(pi.process,10000);}
      if(pi.thread!=IntPtr.Zero)CloseHandle(pi.thread);if(pi.process!=IntPtr.Zero)CloseHandle(pi.process);if(job!=IntPtr.Zero)CloseHandle(job);
      if(attrs!=IntPtr.Zero){DeleteProcThreadAttributeList(attrs);Marshal.FreeHGlobal(attrs);}if(capMem!=IntPtr.Zero)Marshal.FreeHGlobal(capMem);if(handleMem!=IntPtr.Zero)Marshal.FreeHGlobal(handleMem);if(envMem!=IntPtr.Zero)Marshal.FreeHGlobal(envMem);
      foreach(var handle in handles)CloseHandle(handle);
      if(output!=null)output.Dispose();if(error!=null)error.Dispose();
    }
  }
